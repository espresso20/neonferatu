The hierarchy

build/color.py
# the ceiling on how saturated this hue gets, at this lightness
def max_chroma(L: float, H: float) -> float:
    """Walk chroma down until the color fits in sRGB."""
    lo, hi = 0.0, 0.5
    for _ in range(48):
        mid = (lo + hi) / 2
        if in_gamut(L, mid, H):
            lo = mid
        else:
            hi = mid
    return lo
